cmake_minimum_required(VERSION 3.13)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(pico_pio_encoder_pull_tests C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
pico_sdk_init()
add_subdirectory(../.. pico_pio_encoder)
add_executable(pull_modes pull_modes.cpp)
target_link_libraries(pull_modes pico_stdlib pico_pio_encoder)
option(PICO_PIO_ENCODER_NIDEC_FIXTURE "Hold the documented Nidec fixture stopped" OFF)
if(PICO_PIO_ENCODER_NIDEC_FIXTURE)
  target_compile_definitions(pull_modes PRIVATE PICO_PIO_ENCODER_NIDEC_FIXTURE=1)
endif()
pico_enable_stdio_usb(pull_modes 1)
pico_enable_stdio_uart(pull_modes 0)
pico_add_extra_outputs(pull_modes)
